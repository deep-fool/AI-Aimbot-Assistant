#include <Mouse.h>
#include <math.h>

const int CENTER_X = 1280; // 请确保这与你游戏的实际中心分辨率一致
const int CENTER_Y = 800;
const float IMG_WIDTH = 2560.0;
const float IMG_HEIGHT = 1600.0; 

// ==========================================
// 核心：纯净版自适应 PID (去除了多余的耦合)
// ==========================================
class AdaptivePID {
private:
    float kp, ki, kd;
    float integral = 0.0;
    float last_error = 0.0;
    int stable_count = 0;
    bool is_reached = false;
    
    // 底层状态机参数 (根据你的 C++ 逻辑重构)
    float reach_thresh = 5.0;       // 达标误差阈值 (像素)
    float min_coef = 1.0;           // 最小系数 
    float max_coef = 3.0;           // 最大系数 
    float sharpness = 5.0;          // 过渡锐度
    float midpoint = 0.1;           // 动态过渡中点 (默认目标占比 10%)
    int min_data = 2;               // 最小数据量
    int error_tol = 3;              // 误差变化容限
    float smooth_factor = 0.6;      // 平滑因子 (类似低通滤波，0.1~1.0，越小越平滑)
    
public:
    AdaptivePID(float p, float i, float d) : kp(p), ki(i), kd(d) {}
    
    float update(float error, float dt, float target_w, float img_size) {
        // --- 动态阈值计算 (Sigmoid) ---
        float w_ratio = target_w / img_size;
        float dynamic_coef = min_coef + (max_coef - min_coef) / (1.0f + exp(-sharpness * (w_ratio - midpoint)));
        float dynamic_thresh = dynamic_coef * target_w;

        // --- 状态机判定 ---
        if (!is_reached && abs(error) < reach_thresh) {
            is_reached = true;
        } 
        else if (abs(error) >= dynamic_thresh) {
            is_reached = false;
            integral = 0; // 严重偏离，清空积分防止“画圈”或甩过头
            stable_count = 0;
        } 
        else if (!is_reached && abs(error) >= reach_thresh && abs(error) <= dynamic_thresh) {
            float diff = abs(error - last_error);
            if (diff < error_tol) stable_count++;
            else stable_count = 0;
            
            if (stable_count >= min_data) {
                is_reached = true; 
                stable_count = 0;
                integral = 0; 
            }
        }

        // --- PID 核心计算 ---
        // 注意：鼠标移动是离散的，且 dt 通常极小(0.01s)，会导致导数极大，所以这套系统的 kd 参数必须极小。
        float err_rate = (error - last_error) / dt;
        float p_out, i_out, d_out;

        if (is_reached) {
            integral += error * dt;
            // 积分限幅 (Anti-windup)，极其重要！防止鼠标粘滞
            integral = constrain(integral, -100.0f, 100.0f);
            
            i_out = ki * integral;
            p_out = kp * error;
            d_out = kd * err_rate;
        } else {
            // 未达标时，削弱移动力度，防止甩头
            integral += (error * 0.5f) * dt;
            integral = constrain(integral, -100.0f, 100.0f);
            
            i_out = ki * integral;
            p_out = (kp * 0.5f) * error;     
            d_out = kd * err_rate;
        }

        float raw_out = p_out + i_out + d_out;
        float total_out = smooth_factor * raw_out; // 最终平滑输出

        last_error = error;
        return total_out;
    }

    void reset() {
        integral = 0;
        last_error = 0;
        stable_count = 0;
        is_reached = false;
    }
};

// ==========================================
// 实例化 PID 控制器 (绝对安全的起步参数)
// ==========================================
// 参数解析:
// Kp = 0.35 : 决定拉枪速度。像素误差100时，每帧走约35像素。
// Ki = 0.0  : Aimbot 通常不需要积分项，除非你发现它永远瞄准在敌人边缘。先设为0。
// Kd = 0.002: 阻尼器。因为除以了很小的 dt，这个值稍微大一点就会死锁不走。
AdaptivePID pidX(0.1, 0.0, 0.0015); 
AdaptivePID pidY(0.1, 0.0, 0.0015);

// 拟人化延迟参数 (随机防封)
int min_delay = 1;
int max_delay = 3;

unsigned long last_move_time = 0;
unsigned long last_pid_time = 0;

void setup() {
  Serial.begin(115200);
  Serial.setTimeout(1); // 提高串口读取速度，极度重要
  Mouse.begin();
  randomSeed(analogRead(0));  
  last_pid_time = millis();
}

void loop() {
  if (Serial.available() > 0) {
    int targetX = Serial.parseInt();
    int targetY = Serial.parseInt();
    int val3 = Serial.parseInt();
    int val4 = Serial.parseInt();

    int targetW = 80; // 默认宽度
    int isfire = 0;

    // 智能解析 3位或4位数据
    if (val4 == 0 && (val3 == 0 || val3 == 1)) {
        isfire = val3;
    } else {
        targetW = val3;
        isfire = val4;
        if (targetW < 10) targetW = 10; 
    }

    // 异常坐标过滤
    if (targetX < 100 || targetY < 100 || targetX > 3000 || targetY > 2000) return;

    float error_x = targetX - CENTER_X;
    float error_y = targetY - CENTER_Y;

    // --- 严苛的死区判定 (绝对静止) ---
    // 如果已经在准心周围 3 像素以内，完全停止一切计算，直接退出。
    // 这能彻底消灭“准星中心微抖”。
    if (abs(error_x) <= 3 && abs(error_y) <= 3) {
      pidX.reset(); 
      pidY.reset();
      
      // 如果目标在准心内且上位机要求开火
      if (isfire == 1) {
        Mouse.press(MOUSE_LEFT);
        delay(2); 
        Mouse.release(MOUSE_LEFT);
      }
      return; 
    }

    // 计算精准 dt (秒)
    unsigned long now = millis();
    float dt = (now - last_pid_time) / 1000.0f;
    if (dt <= 0.0f) dt = 0.001f; // 防除零
    last_pid_time = now;

    // --- 获取 PID 计算的纯净移动步长 ---
    float move_step_x = pidX.update(error_x, dt, targetW, IMG_WIDTH);
    float move_step_y = pidY.update(error_y, dt, targetW, IMG_HEIGHT);

    // 限制单次移动物理极限，防止瞬间瞬移导致画面撕裂或触发反作弊
    int final_move_x = constrain((int)round(move_step_x), -100, 100);
    int final_move_y = constrain((int)round(move_step_y), -100, 100);

    // 发送给鼠标硬件
    if (millis() - last_move_time > 5) {
      delay(random(min_delay, max_delay + 1));
      
      // 只有在真的需要移动时才调用硬件，减少 USB 拥堵
      if (final_move_x != 0 || final_move_y != 0) {
        Mouse.move(final_move_x, final_move_y, 0);
      }
      
      last_move_time = millis();
    }

    // 清理废弃字符，保持流顺畅
    while (Serial.available() > 0 && isspace(Serial.peek())) Serial.read();
  }
}
